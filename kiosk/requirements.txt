# Core UI
# 2.3.1, not 2.3.0: 2.3.0 doesn't build under Python 3.13 (its Cython-generated
# _event.c uses C API removed in 3.13); 2.3.1 added 3.13 support.
kivy==2.3.1

# Cloud backend
supabase==2.31.0
httpx==0.27.2
# realtime==2.31.0 (pulled in by supabase) imports websockets.asyncio,
# which only exists from websockets 13 onward — but realtime's own
# declared range (>=11,<16) doesn't guarantee that, so an unpinned
# install can still resolve websockets 12.x and crash every Supabase
# client build with "No module named 'websockets.asyncio'".
websockets>=13,<16
python-dotenv==1.0.1

# Note: Realtime confirmation (services/supabase_client.py
# RealtimeConfirmer) will still log a harmless warning and no-op even
# with these versions — supabase-py's *sync* client (what this kiosk
# uses everywhere) explicitly doesn't support Realtime; only the async
# client does. This is a known, accepted limitation: the REST insert
# result already gives the kiosk a reliable, synchronous success/failure
# signal, which is what actually drives the upload UI.

# Hardware: sensors and fan live on the ESP32, reached over HTTP
# (hardware/esp32.py). No GPIO/SPI libraries are needed on the Pi any more.
requests==2.34.2
